1. Postgres accepts TLS-only connections (self-signed cert baked into the
   image) and requires the POSTGRES_PASSWORD set in this release's Secret
   (default "changeme" - override postgresPassword or existingSecretName
   before using this for anything beyond a quick trial).

2. Connect from inside the cluster:
   {{ include "postgres.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.port }}

3. To connect from your machine:
   export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "postgres.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
   kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 5432:5432
   kubectl --namespace {{ .Release.Namespace }} exec -it $POD_NAME -- env PGSSLMODE=require psql -U {{ .Values.postgresUser }} -d {{ .Values.postgresDb }}
